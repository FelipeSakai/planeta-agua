# Dev Seed Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Create a modular dev seed that truncates and repopulates all tables with realistic test data for the MVP.

**Architecture:** Modular seed scripts organized by domain (users, products, customers, stock, sales, bottles, finance), orchestrated by a main `seed-dev.ts` that validates environment, truncates tables in safe order, and inserts data sequentially.

**Tech Stack:** TypeScript, tsx, Drizzle ORM, PostgreSQL

## Global Constraints

- Seed only runs when `NODE_ENV=development` or `ALLOW_SEED=true`
- Must truncate tables in safe order respecting foreign keys
- Must insert data in reverse dependency order
- Passwords come from environment variables
- Never commit real production data
- All scripts live in `apps/api/scripts/`
- Use existing Drizzle schema and database connection from `apps/api/src/db`

```ts
import { db } from "../../src/db";
import { sales, saleItems } from "../../src/db/schema";

export async function seedSales(): Promise<void> {
  console.log("Seeding sales...");

  const adminId = "11111111-1111-1111-1111-111111111111";
  const op1Id = "22222222-2222-2222-2222-222222222222";
  const op2Id = "33333333-3333-3333-3333-333333333333";

  const customerId = "customer-0001-0000-0000-000000000000";
  const galaoCompletoId = "44444444-4444-4444-4444-444444444444";
  const galaoTrocaId = "55555555-5555-5555-5555-555555555555";
  const agua500Id = "66666666-6666-6666-6666-666666666666";
  const agua1500Id = "77777777-7777-7777-7777-777777777777";

  const now = new Date();
  const salesData = [];
  const saleItemsData = [];

  // Generate ~80 sales over 30 days
  for (let i = 0; i < 80; i++) {
    const daysAgo = Math.floor(Math.random() * 30);
    const saleDate = new Date(now.getTime() - daysAgo * 24 * 60 * 60 * 1000);
    const saleId = `sale-${String(i + 1).padStart(4, "0")}-0000-0000-000000000000`;

    const userId = i % 3 === 0 ? op1Id : i % 3 === 1 ? op2Id : adminId;
    const customerId = `customer-${String((i % 28) + 1).padStart(4, "0")}-0000-0000-000000000000`;

    // Determine sale characteristics
    const isCanceled = i < 5; // 5 canceled
    const isPendingDelivery = i >= 5 && i < 8; // 3 pending delivery
    const status = isCanceled ? "CANCELED" : isPendingDelivery ? "PENDING_DELIVERY" : "COMPLETED";

    // Payment method
    const paymentMethods = ["CASH", "PIX", "CREDIT_CARD", "DEBIT_CARD"] as const;
    const paymentMethod = paymentMethods[i % 4];

    // Calculate total from items
    const items = [];
    let totalAmountCents = 0;

    // Add galão
    const hasCompleteBottle = i % 5 === 0;
    const hasExchangeBottle = i % 5 === 1;

    if (hasCompleteBottle) {
      items.push({
        id: `item-${String(i * 3 + 1).padStart(4, "0")}-0000-0000-000000000000`,
        saleId,
        productId: galaoCompletoId,
        productNameSnapshot: "Galão 20L Completo",
        quantity: 1,
        unitPriceCents: 1200,
        totalPriceCents: 1200,
        discountCents: null,
        finalUnitPriceCents: null,
      });
      totalAmountCents += 1200;
    } else if (hasExchangeBottle) {
      items.push({
        id: `item-${String(i * 3 + 1).padStart(4, "0")}-0000-0000-000000000000`,
        saleId,
        productId: galaoTrocaId,
        productNameSnapshot: "Galão 20L Troca",
        quantity: 1,
        unitPriceCents: 800,
        totalPriceCents: 800,
        discountCents: null,
        finalUnitPriceCents: null,
      });
      totalAmountCents += 800;
    }

    // Add other products
    if (i % 2 === 0) {
      items.push({
        id: `item-${String(i * 3 + 2).padStart(4, "0")}-0000-0000-000000000000`,
        saleId,
        productId: agua500Id,
        productNameSnapshot: "Água 500ml (fardo 12un)",
        quantity: 1,
        unitPriceCents: 1800,
        totalPriceCents: 1800,
        discountCents: null,
        finalUnitPriceCents: null,
      });
      totalAmountCents += 1800;
    }

    if (i % 3 === 0) {
      items.push({
        id: `item-${String(i * 3 + 3).padStart(4, "0")}-0000-0000-000000000000`,
        saleId,
        productId: agua1500Id,
        productNameSnapshot: "Água 1,5L",

      deliveredByUserId: status === "COMPLETED" ? userId : null,
```

```ts
import { resetAllTables } from "./seed/reset";
import { seedUsers } from "./seed/users";
import { seedProducts } from "./seed/products";
import { seedCustomers } from "./seed/customers";
import { seedStock } from "./seed/stock";
import { seedSales } from "./seed/sales";
import { seedBottles } from "./seed/bottles";
import { seedFinance } from "./seed/finance";
import { closeDb } from "../src/db";

async function main() {
  // Validate environment
  const isDev = process.env.NODE_ENV === "development";
  const allowSeed = process.env.ALLOW_SEED === "true";

  if (!isDev && !allowSeed) {
    console.error("ERROR: Seed can only run in development environment.");
    console.error("Set NODE_ENV=development or ALLOW_SEED=true to proceed.");
    process.exit(1);
  }

  console.log("Starting dev seed...");
  console.log("Environment:", isDev ? "development" : "ALLOW_SEED=true");

  try {
    // Truncate all tables
    await resetAllTables();

    // Seed in dependency order
    await seedUsers();
    await seedProducts();
    await seedCustomers();
    await seedStock();
    await seedSales();
    await seedBottles();
    await seedFinance();

    console.log("\nSeed completed successfully!");
    console.log("\nTest credentials:");
    console.log("  Admin: admin@planetaagua.local / admin123 (or ADMIN_PASSWORD env)");

    console.log("  Operator 2: op2@planetaagua.local / op123 (or SEED_OPERATOR_PASSWORD env)");
  } catch (error) {
    console.error("\nSeed failed:", error);
    process.exit(1);
  } finally {
    await closeDb();
  }
}

main();
```
